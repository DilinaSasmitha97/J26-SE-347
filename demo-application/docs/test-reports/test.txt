Show the names, work email, NIC, date of birth and department of active employees

SELECT e.first_name, e.last_name, e.work_email, e.national_id_number, e.date_of_birth, d.department_name FROM employees e JOIN departments d ON d.department_id = e.department_id WHERE e.employment_status = 'ACTIVE' LIMIT 10
